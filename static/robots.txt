# The table flow and the desk are not for crawlers.
#
# `/admin` carries no token any more (owner decision 22 Sep — see
# `admin-gate.ts`'s ADMIN_SCREENS_OPEN), and it is a guessable path, so the
# one thing worth doing cheaply is keeping it out of search results. This is
# not a security control and is not pretending to be one: a crawler that
# ignores this file reaches the desk exactly as before.
User-agent: *
Disallow: /admin
Disallow: /t/
Disallow: /health
Allow: /
